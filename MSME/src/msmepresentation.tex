% SLIDE 2: CLINICAL PATHOLOGY & PROBLEM (WHAT IS SEPSIS?)
% =========================================================================================
\begin{frame}{Clinical Pathology: What is Sepsis \& Why Time Matters}
    \begin{columns}[t]
        \begin{column}{0.48\textwidth}
            \begin{block}{\textcolor{neonRose}{Clinical Definition of Sepsis}}
                \scriptsize
                \begin{itemize}
                    \item \textbf{Dysregulated Immune Response:} Life-threatening organ dysfunction caused by a severe, systemic immune reaction to infection.
                    \item \textbf{Progression to Shock:} Causes capillary leakage, tissue hypoxia, Multi-Organ Dysfunction Syndrome (MODS), and rapid circulatory collapse.
                    \item \textbf{The Golden Window:} Patient survival drops by \textbf{8\% for every hour} diagnosis and antimicrobial treatment are delayed.
                \end{itemize}
            \end{block}
        \end{column}
        
        \begin{column}{0.48\textwidth}
            \begin{block}{\textcolor{neonAmber}{Why Current ICU Alarms Fail}}
                \scriptsize
                \begin{itemize}
                    \item \textbf{Static Threshold Alarms:} Legacy monitors trigger alerts only after vitals breach rigid limits (e.g., HR > 100).
                    \item \textbf{Severe Alarm Fatigue:} \textbf{85\%--95\%} of bedside alarms are false positives, causing clinicians to desensitize.
                    \item \textbf{Missed Temporal Signals:} Fails to detect complex, multi-vital co-dependencies occurring hours prior to septic crash.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    
    \vspace{0.1cm}
    \begin{alertblock}{\textcolor{neonCyan}{Indian Healthcare Mortality Impact}}
        \scriptsize
        Sepsis accounts for \textbf{3.0 Million deaths annually in India} (\textbf{30\%} of total hospital deaths), making early automated crash prediction an urgent national healthcare priority.
    \end{alertblock}
\end{frame}
