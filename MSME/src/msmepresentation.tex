        \begin{column}{0.48\textwidth}
            \begin{block}{\textcolor{neonAmber}{3. Client-Side Selective Personalization}}
                \scriptsize

                \begin{itemize}
                    \item \textbf{38\% Bandwidth Savings} (1.52 GB vs 2.48 GB).
                    \item \textbf{Rapid Calibration:} Adapts in \textbf{6 minutes}.
                \end{itemize}
            \end{block}

            \begin{block}{\textcolor{neonRose}{4. Temporal XAI Saliency Heatmaps}}
